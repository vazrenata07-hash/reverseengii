cmake_minimum_required(VERSION 3.22)
project(ycidentity LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT DEFINED YC_VENDOR)
    message(FATAL_ERROR "YC_VENDOR must point to the pinned dependency sources")
endif()
set(LSPLANT_STANDALONE OFF CACHE BOOL "" FORCE)
set(LSPLANT_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(DOBBY_DEBUG OFF CACHE BOOL "" FORCE)
set(DOBBY_GENERATE_SHARED OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(BUILD_TEST OFF CACHE BOOL "" FORCE)
set(Plugin.SymbolResolver OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_THREADS OFF CACHE STRING "" FORCE)
set(ENCODERS "" CACHE STRING "" FORCE)
set(MICROLZMA_ENCODER OFF CACHE BOOL "" FORCE)
set(MICROLZMA_DECODER OFF CACHE BOOL "" FORCE)
set(LZIP_DECODER OFF CACHE BOOL "" FORCE)
add_subdirectory("${YC_VENDOR}/lsplant/lsplant/src/main/jni" lsplant EXCLUDE_FROM_ALL)
add_subdirectory("${YC_VENDOR}/dobby" dobby EXCLUDE_FROM_ALL)
add_subdirectory("${YC_VENDOR}/xz" xz EXCLUDE_FROM_ALL)
add_library(ycidentity SHARED bridge.cpp art_symbols.cpp)
target_include_directories(ycidentity PRIVATE "${YC_VENDOR}/xz/src/liblzma/api")
target_compile_options(ycidentity PRIVATE -Wall -Wextra -fvisibility=hidden -ffunction-sections -fdata-sections)
target_link_options(ycidentity PRIVATE -Wl,--gc-sections -Wl,--exclude-libs,ALL)
target_link_libraries(ycidentity PRIVATE lsplant_static dobby liblzma log dl)
